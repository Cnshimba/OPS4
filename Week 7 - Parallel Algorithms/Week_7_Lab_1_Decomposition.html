📋 Overview

A production HPC system is more than a raw MPI compiler — it is an organised scientific ecosystem. This lab guides you through building a modular scientific software stack: optimised linear algebra libraries, parallel Fourier solvers, hierarchical scientific databases, and high-level scientific toolkits from source, organised under Lmod environment modules.

Section 10: Verification – Parallel Monte Carlo Pi Estimation

Create /shared/mpi_pi.c — a parallel Monte Carlo Pi estimator using MPI_Bcast and MPI_Reduce:

Terminal
Terminal Command 
#include <mpi.h>
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main(int argc, char **argv) {
    int rank, size;
    long long total_trials = 1000000000LL;
    long long local_trials, local_hits = 0, global_hits = 0;
    double x, y, pi_estimate, error;
    unsigned int seed;

    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    MPI_Bcast(&total_trials, 1, MPI_LONG_LONG, 0, MPI_COMM_WORLD);
    local_trials = total_trials / size;
    seed = time(NULL) + rank * 1000;

    for (long long i = 0; i < local_trials; i++) {
        x = (double)rand_r(&seed) / RAND_MAX;
        y = (double)rand_r(&seed) / RAND_MAX;
        if (x * x + y * y <= 1.0) local_hits++;
    }

    MPI_Reduce(&local_hits, &global_hits, 1, MPI_LONG_LONG, MPI_SUM, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        pi_estimate = 4.0 * (double)global_hits / total_trials;
        error = pi_estimate - 3.141592653589793;
        printf("=== MPI Collective Pi Estimation Result ===\n");
        printf("Ranks Coordinating : %d\n", size);
        printf("Total Cluster Trials: %lld\n", total_trials);
        printf("Estimated Pi Value : %.10f\n", pi_estimate);
        printf("Calculated Error   : %.10f\n", error);
        printf("===========================================\n");
    }
    MPI_Finalize();
    return 0;
}
Terminal
Terminal Command

srun -N 4 --ntasks-per-node=2 /shared/mpi_pi
